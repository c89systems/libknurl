cmake_minimum_required(VERSION 3.24)
project(knurl_example LANGUAGES C CXX)
set(KNURL_VARIANT shared CACHE STRING "static or shared")
set(KNURL_DISCOVERY cmake CACHE STRING "cmake or pkgconfig")
if(KNURL_VERSION)
  set(knurl_version_args ${KNURL_VERSION} EXACT)
  set(knurl_pc_version "=${KNURL_VERSION}")
endif()
if(NOT KNURL_VARIANT MATCHES "^(static|shared)$" OR
    NOT KNURL_DISCOVERY MATCHES "^(cmake|pkgconfig)$")
  message(FATAL_ERROR "Select static|shared and cmake|pkgconfig")
endif()
foreach(prefix "${KNURL_PREFIX}" "${KNURL_SDK_ROOT}/cpkt"
    "${KNURL_SDK_ROOT}/lonejson" "${KNURL_SDK_ROOT}/pslog")
  list(APPEND CMAKE_PREFIX_PATH "${prefix}")
  list(APPEND CMAKE_FIND_ROOT_PATH "${prefix}")
endforeach()
add_executable(server server.c)
set_target_properties(server PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(server PRIVATE -std=c89 -Wall -Wextra -Wpedantic -Werror)
if(KNURL_DISCOVERY STREQUAL "cmake")
  find_package(knurl ${knurl_version_args} CONFIG REQUIRED COMPONENTS ${KNURL_VARIANT})
  target_link_libraries(server PRIVATE knurl::${KNURL_VARIANT})
else()
  find_package(PkgConfig REQUIRED)
  if(KNURL_VARIANT STREQUAL "static")
    set(PKG_CONFIG_ARGN --static)
  endif()
  pkg_check_modules(KNURL REQUIRED "knurl${knurl_pc_version}")
  target_include_directories(server PRIVATE ${KNURL_INCLUDE_DIRS})
  target_compile_options(server PRIVATE ${KNURL_CFLAGS_OTHER})
  target_link_directories(server PRIVATE ${KNURL_LIBRARY_DIRS})
  target_link_options(server PRIVATE ${KNURL_LDFLAGS_OTHER})
  set(libraries ${KNURL_LINK_LIBRARIES})
  if(KNURL_VARIANT STREQUAL "static")
    list(FILTER libraries EXCLUDE REGEX "(^knurl$|/libknurl[.])")
    set(static_libraries "")
    foreach(library IN LISTS libraries)
      string(REGEX REPLACE "([.]so([.][0-9]+)*|([.][0-9]+)*[.]dylib)$" ".a" archive "${library}")
      if(NOT archive STREQUAL library AND EXISTS "${archive}")
        list(APPEND static_libraries "${archive}")
      else()
        list(APPEND static_libraries "${library}")
      endif()
    endforeach()
    set(libraries ${static_libraries})
    target_link_libraries(server PRIVATE "${KNURL_PREFIX}/lib/libknurl.a")
  endif()
  target_link_libraries(server PRIVATE ${libraries})
endif()
# Rehearsal runners use the selected toolchain loader and external provider
# SDKs. This is build-tree execution metadata, never installed in libknurl.
if(KNURL_VERIFY_RUN AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  file(GLOB loaders "${CMAKE_SYSROOT}/lib/ld-linux*.so*" "${CMAKE_SYSROOT}/lib/ld-musl*.so*")
  list(GET loaders 0 loader)
  target_link_options(server PRIVATE "-Wl,--dynamic-linker,${loader}" -Wl,--disable-new-dtags)
  set_target_properties(server PROPERTIES BUILD_RPATH
    "${KNURL_PREFIX}/lib;${KNURL_SDK_ROOT}/cpkt/lib;${KNURL_SDK_ROOT}/lonejson/lib;${KNURL_SDK_ROOT}/pslog/lib;${CMAKE_SYSROOT}/lib;${CMAKE_SYSROOT}/usr/lib")
endif()
