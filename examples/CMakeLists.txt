cmake_minimum_required(VERSION 3.24)
project(rotmos_example LANGUAGES C CXX)
set(ROTMOS_VARIANT shared CACHE STRING "static or shared")
set(ROTMOS_DISCOVERY cmake CACHE STRING "cmake or pkgconfig")
if(ROTMOS_VERSION)
  set(rotmos_version_args ${ROTMOS_VERSION} EXACT)
  set(rotmos_pc_version "=${ROTMOS_VERSION}")
endif()
if(NOT ROTMOS_VARIANT MATCHES "^(static|shared)$" OR
    NOT ROTMOS_DISCOVERY MATCHES "^(cmake|pkgconfig)$")
  message(FATAL_ERROR "Select static|shared and cmake|pkgconfig")
endif()
foreach(prefix "${ROTMOS_PREFIX}" "${ROTMOS_SDK_ROOT}/cpkt"
    "${ROTMOS_SDK_ROOT}/lonejson" "${ROTMOS_SDK_ROOT}/pslog")
  list(APPEND CMAKE_PREFIX_PATH "${prefix}")
  list(APPEND CMAKE_FIND_ROOT_PATH "${prefix}")
endforeach()
add_executable(server server.c)
set_target_properties(server PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(server PRIVATE -std=c89 -Wall -Wextra -Wpedantic -Werror)
if(ROTMOS_DISCOVERY STREQUAL "cmake")
  find_package(rotmos ${rotmos_version_args} CONFIG REQUIRED COMPONENTS ${ROTMOS_VARIANT})
  target_link_libraries(server PRIVATE rotmos::${ROTMOS_VARIANT})
else()
  find_package(PkgConfig REQUIRED)
  if(ROTMOS_VARIANT STREQUAL "static")
    set(PKG_CONFIG_ARGN --static)
  endif()
  pkg_check_modules(ROTMOS REQUIRED "rotmos${rotmos_pc_version}")
  target_include_directories(server PRIVATE ${ROTMOS_INCLUDE_DIRS})
  target_compile_options(server PRIVATE ${ROTMOS_CFLAGS_OTHER})
  target_link_directories(server PRIVATE ${ROTMOS_LIBRARY_DIRS})
  target_link_options(server PRIVATE ${ROTMOS_LDFLAGS_OTHER})
  set(libraries ${ROTMOS_LINK_LIBRARIES})
  if(ROTMOS_VARIANT STREQUAL "static")
    list(FILTER libraries EXCLUDE REGEX "(^rotmos$|/librotmos[.])")
    set(static_libraries "")
    foreach(library IN LISTS libraries)
      string(REGEX REPLACE "([.]so([.][0-9]+)*|([.][0-9]+)*[.]dylib)$" ".a" archive "${library}")
      if(NOT archive STREQUAL library AND EXISTS "${archive}")
        list(APPEND static_libraries "${archive}")
      else()
        list(APPEND static_libraries "${library}")
      endif()
    endforeach()
    set(libraries ${static_libraries})
    target_link_libraries(server PRIVATE "${ROTMOS_PREFIX}/lib/librotmos.a")
  endif()
  target_link_libraries(server PRIVATE ${libraries})
endif()
# Rehearsal runners use the selected toolchain loader and external provider
# SDKs. This is build-tree execution metadata, never installed in librotmos.
if(ROTMOS_VERIFY_RUN AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  file(GLOB loaders "${CMAKE_SYSROOT}/lib/ld-linux*.so*" "${CMAKE_SYSROOT}/lib/ld-musl*.so*")
  list(GET loaders 0 loader)
  target_link_options(server PRIVATE "-Wl,--dynamic-linker,${loader}" -Wl,--disable-new-dtags)
  set_target_properties(server PROPERTIES BUILD_RPATH
    "${ROTMOS_PREFIX}/lib;${ROTMOS_SDK_ROOT}/cpkt/lib;${ROTMOS_SDK_ROOT}/lonejson/lib;${ROTMOS_SDK_ROOT}/pslog/lib;${CMAKE_SYSROOT}/lib;${CMAKE_SYSROOT}/usr/lib")
endif()
