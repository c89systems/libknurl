cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_TOOLCHAIN_FILE OR NOT KNURL_TARGET)
  message(FATAL_ERROR "Select the lifecycle toolchain and concrete KNURL_TARGET; use make fuzz-build")
endif()
set(KNURL_AFL TRUE CACHE BOOL "Pinned GCC-plugin instrumentation" FORCE)
include(../../cmake/Version.cmake)
get_filename_component(root "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
knurl_version("${root}" version)
project(knurl_fuzz VERSION "${version}" LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
enable_testing()
include("${root}/cmake/OwnedPaths.cmake")
knurl_require_owned_path("${CMAKE_BINARY_DIR}" "Fuzzer build")
foreach(prefix "${KNURL_CORE_PREFIX}" "${KNURL_SDK_ROOT}/cpkt" "${KNURL_SDK_ROOT}/lonejson" "${KNURL_SDK_ROOT}/pslog")
  list(APPEND CMAKE_PREFIX_PATH "${prefix}")
  list(APPEND CMAKE_FIND_ROOT_PATH "${prefix}")
endforeach()
find_package(knurl ${version} EXACT CONFIG REQUIRED COMPONENTS static)
# Instrument the owned parsers, reusing the ordinary installed SDK for support
# operations. Do not rebuild/instrument Kore or dependency SDKs for this lane.
add_executable(knurl_fuzz parsers.c
  "${root}/src/proxy_framing.c" "${root}/src/proxy_headers.c"
  "${root}/src/proxy_http.c" "${root}/src/proxy_ws_wire.c"
  "${root}/src/static_policy.c" "${root}/src/route_path.c"
  "${root}/src/websocket_frame.c" "${root}/src/websocket.c")
target_include_directories(knurl_fuzz PRIVATE "${root}/src")
target_link_libraries(knurl_fuzz PRIVATE knurl::static)
set_target_properties(knurl_fuzz PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(knurl_fuzz PRIVATE -std=c89 -Wall -Wextra -Wpedantic -Werror -Wstrict-prototypes -Wmissing-prototypes -UNDEBUG)
file(GLOB loaders "${CMAKE_SYSROOT}/lib/ld-linux*.so*")
list(GET loaders 0 loader)
target_link_options(knurl_fuzz PRIVATE "-Wl,--dynamic-linker,${loader}" -Wl,--disable-new-dtags)
set_target_properties(knurl_fuzz PROPERTIES BUILD_RPATH "${CMAKE_SYSROOT}/lib;${CMAKE_SYSROOT}/usr/lib;${KNURL_SDK_ROOT}/cpkt/lib;${KNURL_SDK_ROOT}/lonejson/lib;${KNURL_SDK_ROOT}/pslog/lib")
file(GLOB seeds "${CMAKE_CURRENT_SOURCE_DIR}/corpus/*")
foreach(seed IN LISTS seeds)
  get_filename_component(name "${seed}" NAME)
  add_test(NAME "fuzz.replay.${name}" COMMAND knurl_fuzz "${seed}")
  set_tests_properties("fuzz.replay.${name}" PROPERTIES TIMEOUT 5)
endforeach()
