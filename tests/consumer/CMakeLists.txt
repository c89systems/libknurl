cmake_minimum_required(VERSION 3.24)
project(rotmos_consumer LANGUAGES C CXX)
list(APPEND CMAKE_PREFIX_PATH "${ROTMOS_PREFIX}")
list(APPEND CMAKE_FIND_ROOT_PATH "${ROTMOS_PREFIX}")
foreach(provider cpkt lonejson pslog)
  list(APPEND CMAKE_PREFIX_PATH "${ROTMOS_PREFIX}/providers/${provider}")
  list(APPEND CMAKE_FIND_ROOT_PATH "${ROTMOS_PREFIX}/providers/${provider}")
endforeach()
find_package(rotmos ${ROTMOS_VERSION} EXACT CONFIG REQUIRED COMPONENTS ${ROTMOS_VARIANT})
foreach(surface foundation reply declarations server-runtime proxy-runtime proxy-websocket json xml xml-mapped xml-runtime json-runtime asset static-runtime storage-runtime storage-filesystem webdav-runtime source-runtime upload-runtime rest-runtime openapi policy-runtime policy-logs-runtime)
  add_executable(${surface}_consumer ${surface}.c)
  set_target_properties(${surface}_consumer PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_compile_options(${surface}_consumer PRIVATE -std=c89 -Wall -Wextra -Wpedantic -Werror)
  target_compile_definitions(${surface}_consumer PRIVATE
    $<$<PLATFORM_ID:Darwin>:_DARWIN_C_SOURCE=1>)
  target_link_libraries(${surface}_consumer PRIVATE rotmos::${ROTMOS_VARIANT})
  if(surface STREQUAL "policy-logs-runtime")
    target_link_libraries(${surface}_consumer PRIVATE pslog::pslog_${ROTMOS_VARIANT})
  endif()
  if(surface STREQUAL "server-runtime" OR surface STREQUAL "xml" OR surface STREQUAL "xml-mapped" OR surface STREQUAL "asset" OR surface STREQUAL "storage-runtime" OR surface STREQUAL "webdav-runtime" OR surface STREQUAL "source-runtime" OR surface STREQUAL "upload-runtime" OR surface STREQUAL "rest-runtime" OR surface STREQUAL "openapi" OR surface STREQUAL "policy-runtime")
    target_link_libraries(${surface}_consumer PRIVATE Threads::Threads)
  endif()
  if(surface STREQUAL "proxy-websocket")
    target_link_libraries(${surface}_consumer PRIVATE OpenSSL::SSL)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    file(GLOB loaders "${CMAKE_SYSROOT}/lib/ld-linux*.so*" "${CMAKE_SYSROOT}/lib/ld-musl*.so*")
    list(GET loaders 0 loader)
    target_link_options(${surface}_consumer PRIVATE "-Wl,--dynamic-linker,${loader}" -Wl,--disable-new-dtags)
    set_target_properties(${surface}_consumer PROPERTIES BUILD_RPATH
      "${ROTMOS_PREFIX}/lib;${ROTMOS_PREFIX}/providers/cpkt/lib;${ROTMOS_PREFIX}/providers/lonejson/lib;${ROTMOS_PREFIX}/providers/pslog/lib;${CMAKE_SYSROOT}/lib;${CMAKE_SYSROOT}/usr/lib")
  endif()
endforeach()
