cmake_minimum_required(VERSION 3.24)
project(rotmos_consumer LANGUAGES C CXX)
list(APPEND CMAKE_PREFIX_PATH "${ROTMOS_PREFIX}")
list(APPEND CMAKE_FIND_ROOT_PATH "${ROTMOS_PREFIX}")
find_package(rotmos ${ROTMOS_VERSION} EXACT CONFIG REQUIRED COMPONENTS ${ROTMOS_VARIANT})
add_executable(consumer foundation.c)
set_target_properties(consumer PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(consumer PRIVATE -std=c89 -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(consumer PRIVATE rotmos::${ROTMOS_VARIANT})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  file(GLOB loaders "${CMAKE_SYSROOT}/lib/ld-linux*.so*" "${CMAKE_SYSROOT}/lib/ld-musl*.so*")
  list(GET loaders 0 loader)
  target_link_options(consumer PRIVATE "-Wl,--dynamic-linker,${loader}" -Wl,--disable-new-dtags)
  set_target_properties(consumer PROPERTIES BUILD_RPATH
    "${ROTMOS_PREFIX}/lib;${CMAKE_SYSROOT}/lib;${CMAKE_SYSROOT}/usr/lib")
endif()
