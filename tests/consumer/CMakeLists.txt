cmake_minimum_required(VERSION 3.24)
project(knurl_consumer LANGUAGES C CXX)
list(APPEND CMAKE_PREFIX_PATH "${KNURL_PREFIX}")
list(APPEND CMAKE_FIND_ROOT_PATH "${KNURL_PREFIX}")
if(NOT KNURL_SDK_ROOT)
  set(KNURL_SDK_ROOT "${KNURL_PREFIX}/providers")
endif()
foreach(provider cpkt lonejson pslog)
  list(APPEND CMAKE_PREFIX_PATH "${KNURL_SDK_ROOT}/${provider}")
  list(APPEND CMAKE_FIND_ROOT_PATH "${KNURL_SDK_ROOT}/${provider}")
endforeach()
find_package(knurl ${KNURL_VERSION} EXACT CONFIG REQUIRED COMPONENTS ${KNURL_VARIANT})
add_executable(vectis-typed_consumer vectis-typed.c vectis-forward.c vectis-route.c)
set_target_properties(vectis-typed_consumer PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(vectis-typed_consumer PRIVATE -std=c89 -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(vectis-typed_consumer PRIVATE $<$<PLATFORM_ID:Darwin>:_DARWIN_C_SOURCE=1>)
target_link_libraries(vectis-typed_consumer PRIVATE knurl::${KNURL_VARIANT})
foreach(surface vectis-filesystem vectis-host-runtime vectis-xml-runtime materialized-output tls-runtime vectis-services-runtime vectis-request-reply vectis-response-head vectis-body-file vectis-route-runtime vectis-config foundation reply declarations server-runtime proxy-runtime proxy-websocket json xml xml-mapped xml-runtime json-runtime asset mime static-runtime storage-filesystem webdav-runtime source-runtime upload-runtime rest-runtime openapi policy-runtime policy-logs-runtime)
  add_executable(${surface}_consumer ${surface}.c)
  if(surface STREQUAL "vectis-filesystem")
    target_link_libraries(${surface}_consumer PRIVATE Threads::Threads)
  endif()
  if(surface STREQUAL "vectis-host-runtime")
    target_sources(${surface}_consumer PRIVATE vectis-host.c vectis-server.c vectis-route.c)
    target_link_libraries(${surface}_consumer PRIVATE pslog::pslog_${KNURL_VARIANT})
  endif()
  if(surface STREQUAL "vectis-xml-runtime")
    target_sources(${surface}_consumer PRIVATE vectis-xml.c vectis-route.c)
    target_link_libraries(${surface}_consumer PRIVATE Threads::Threads)
  endif()
  if(surface STREQUAL "json-runtime" OR surface STREQUAL "xml-runtime")
    target_sources(${surface}_consumer PRIVATE vectis-forward.c vectis-route.c)
  endif()
  if(surface STREQUAL "vectis-config" OR surface STREQUAL "vectis-response-head" OR
      surface STREQUAL "policy-runtime" OR surface STREQUAL "policy-logs-runtime")
    target_sources(${surface}_consumer PRIVATE vectis-server.c)
  endif()
  if(surface STREQUAL "vectis-services-runtime" OR surface STREQUAL "tls-runtime")
    target_sources(${surface}_consumer PRIVATE vectis-services.c)
    target_link_libraries(${surface}_consumer PRIVATE Threads::Threads)
  endif()
  if(surface STREQUAL "openapi")
    target_sources(${surface}_consumer PRIVATE vectis-services.c vectis-route.c)
  endif()
  if(surface STREQUAL "vectis-route-runtime" OR surface STREQUAL "vectis-body-file" OR surface STREQUAL "vectis-request-reply" OR surface STREQUAL "vectis-services-runtime" OR surface STREQUAL "tls-runtime")
    target_sources(${surface}_consumer PRIVATE vectis-route.c)
  endif()
  set_target_properties(${surface}_consumer PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_compile_options(${surface}_consumer PRIVATE -std=c89 -Wall -Wextra -Wpedantic -Werror)
  target_compile_definitions(${surface}_consumer PRIVATE
    $<$<PLATFORM_ID:Darwin>:_DARWIN_C_SOURCE=1>)
  target_link_libraries(${surface}_consumer PRIVATE knurl::${KNURL_VARIANT})
  if(surface STREQUAL "static-runtime")
    target_link_libraries(${surface}_consumer PRIVATE ZLIB::ZLIB)
  endif()
  if(surface STREQUAL "policy-runtime" OR surface STREQUAL "policy-logs-runtime" OR surface STREQUAL "vectis-config" OR surface STREQUAL "vectis-response-head")
    target_link_libraries(${surface}_consumer PRIVATE pslog::pslog_${KNURL_VARIANT})
  endif()
  if(surface STREQUAL "server-runtime" OR surface STREQUAL "xml" OR surface STREQUAL "xml-mapped" OR surface STREQUAL "xml-runtime" OR surface STREQUAL "asset" OR surface STREQUAL "webdav-runtime" OR surface STREQUAL "source-runtime" OR surface STREQUAL "upload-runtime" OR surface STREQUAL "rest-runtime" OR surface STREQUAL "openapi" OR surface STREQUAL "policy-runtime")
    target_link_libraries(${surface}_consumer PRIVATE Threads::Threads)
  endif()
  if(surface STREQUAL "proxy-websocket" OR surface STREQUAL "tls-runtime")
    if(KNURL_VARIANT STREQUAL "shared")
      target_link_libraries(${surface}_consumer PRIVATE cpkt::openssl_ssl_shared)
    else()
      target_link_libraries(${surface}_consumer PRIVATE OpenSSL::SSL)
    endif()
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    file(GLOB loaders "${CMAKE_SYSROOT}/lib/ld-linux*.so*" "${CMAKE_SYSROOT}/lib/ld-musl*.so*")
    list(GET loaders 0 loader)
    target_link_options(${surface}_consumer PRIVATE "-Wl,--dynamic-linker,${loader}" -Wl,--disable-new-dtags)
    set_target_properties(${surface}_consumer PROPERTIES BUILD_RPATH
      "${KNURL_PREFIX}/lib;${KNURL_SDK_ROOT}/cpkt/lib;${KNURL_SDK_ROOT}/lonejson/lib;${KNURL_SDK_ROOT}/pslog/lib;${CMAKE_SYSROOT}/lib;${CMAKE_SYSROOT}/usr/lib")
  endif()
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(vectis-typed_consumer PRIVATE "-Wl,--dynamic-linker,${loader}" -Wl,--disable-new-dtags)
  set_target_properties(vectis-typed_consumer PROPERTIES BUILD_RPATH
    "${KNURL_PREFIX}/lib;${KNURL_SDK_ROOT}/cpkt/lib;${KNURL_SDK_ROOT}/lonejson/lib;${KNURL_SDK_ROOT}/pslog/lib;${CMAKE_SYSROOT}/lib;${CMAKE_SYSROOT}/usr/lib")
endif()
