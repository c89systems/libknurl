cmake_minimum_required(VERSION 3.24)
message(FATAL_ERROR "Archived Knurl executable: deferred until Luthrum; build the C library from the repository root")
if(NOT KNURL_TARGET MATCHES "^(x86_64|aarch64)-linux-musl$" OR NOT CMAKE_TOOLCHAIN_FILE)
  message(FATAL_ERROR "knurl executable supports only x86_64/aarch64 Linux musl; use make knurl")
endif()
include(../cmake/Version.cmake)
knurl_version("${CMAKE_CURRENT_SOURCE_DIR}/.." knurl_version)
project(knurl_executable VERSION "${knurl_version}" LANGUAGES C)
get_filename_component(knurl_root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
include("${knurl_root}/cmake/OwnedPaths.cmake")
knurl_require_owned_path("${CMAKE_BINARY_DIR}" "Executable build")
foreach(name cpkt lonejson pslog pid0)
  list(APPEND CMAKE_PREFIX_PATH "${KNURL_SDK_ROOT}/${name}")
  list(APPEND CMAKE_FIND_ROOT_PATH "${KNURL_SDK_ROOT}/${name}")
endforeach()
list(APPEND CMAKE_PREFIX_PATH "${KNURL_CORE_PREFIX}")
list(APPEND CMAKE_FIND_ROOT_PATH "${KNURL_CORE_PREFIX}")
find_package(knurl ${PROJECT_VERSION} EXACT CONFIG REQUIRED COMPONENTS static)
find_package(Lua 5.5.1 EXACT CONFIG REQUIRED)
find_package(pid0 0.6.0 EXACT CONFIG REQUIRED)
include("${knurl_root}/lua/cmake/Sources.cmake")
set(json_source "${KNURL_LUA_SOURCE_ROOT}/lonejson-lua-0.50.0")
set(log_source "${KNURL_LUA_SOURCE_ROOT}/pslog-lua-0.13.0")
foreach(provider json log)
  knurl_require_owned_path("${${provider}_source}" "Released Lua source")
endforeach()
foreach(pair "${json_source}/include/lonejson.h|${KNURL_SDK_ROOT}/lonejson/include/lonejson.h"
    "${log_source}/include/pslog.h|${KNURL_SDK_ROOT}/pslog/include/pslog.h")
  string(REPLACE "|" ";" files "${pair}")
  list(GET files 0 a)
  list(GET files 1 b)
  file(SHA256 "${a}" ah)
  file(SHA256 "${b}" bh)
  if(NOT ah STREQUAL bh)
    message(FATAL_ERROR "Released Lua binding header does not match its SDK")
  endif()
endforeach()
add_library(knurl_embedded_providers STATIC "${json_source}/src/lua/lonejson_lua.c" "${log_source}/lua/src/pslog_lua.c")
target_include_directories(knurl_embedded_providers PRIVATE "${log_source}/include")
target_compile_definitions(knurl_embedded_providers PRIVATE inline= _GNU_SOURCE=1)
target_compile_options(knurl_embedded_providers PRIVATE -std=c89 -Wall -Wextra -Werror)
target_link_libraries(knurl_embedded_providers PUBLIC lonejson::lonejson_static pslog::pslog_static Lua::Lua)
foreach(entry "startup|${CMAKE_CURRENT_SOURCE_DIR}/startup.lua" "facade|${knurl_root}/lua/knurl.lua" "curl_facade|${knurl_root}/lua/curl.lua"
    "lonejson_facade|${json_source}/lua/lonejson/init.lua" "pslog_facade|${log_source}/lua/pslog/init.lua")
  string(REPLACE "|" ";" fields "${entry}")
  list(GET fields 0 name)
  list(GET fields 1 input)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${name}.inc"
    COMMAND "${CMAKE_COMMAND}" "-DINPUT=${input}" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/${name}.inc"
      -P "${knurl_root}/lua/cmake/Embed.cmake" DEPENDS "${input}" "${knurl_root}/lua/cmake/Embed.cmake" VERBATIM)
  list(APPEND embedded_sources "${CMAKE_CURRENT_BINARY_DIR}/${name}.inc")
endforeach()
foreach(name static proxy webdav combined)
  set(input "${CMAKE_CURRENT_SOURCE_DIR}/examples/knurl-example-${name}.lua")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/example_${name}.inc"
    COMMAND "${CMAKE_COMMAND}" "-DINPUT=${input}" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/example_${name}.inc"
      -P "${knurl_root}/lua/cmake/Embed.cmake" DEPENDS "${input}" "${knurl_root}/lua/cmake/Embed.cmake" VERBATIM)
  list(APPEND embedded_sources "${CMAKE_CURRENT_BINARY_DIR}/example_${name}.inc")
endforeach()
add_executable(knurl main.c examples.c ${knurl_lua_sources} ${embedded_sources})
target_include_directories(knurl PRIVATE "${knurl_root}/lua/include" "${CMAKE_CURRENT_BINARY_DIR}"
  "${json_source}/src/lua" "${log_source}/include")
target_link_libraries(knurl PRIVATE knurl::static knurl_embedded_providers CURL::libcurl pid0::pid0_static)
target_compile_options(knurl PRIVATE -std=c89 -Wall -Wextra -Wpedantic -Werror -Wstrict-prototypes -Wmissing-prototypes -Wvla -Walloca)
foreach(target knurl knurl_embedded_providers)
  target_compile_options(${target} PRIVATE "-ffile-prefix-map=${knurl_root}=."
    "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=build")
endforeach()
target_compile_definitions(knurl PRIVATE _GNU_SOURCE=1 "KNURL_EXECUTABLE_VERSION=\"${PROJECT_VERSION}\"")
target_link_options(knurl PRIVATE -static -Wl,--no-undefined)
set_target_properties(knurl PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
install(TARGETS knurl RUNTIME DESTINATION bin)
install(FILES knurl.1 DESTINATION share/man/man1)
install(FILES README.md "${knurl_root}/LICENSE" DESTINATION share/doc/knurl)
install(DIRECTORY examples/ DESTINATION share/doc/knurl/examples)
include("${knurl_root}/cmake/PublicDocs.cmake")
foreach(path IN LISTS knurl_public_docs)
  get_filename_component(directory "${path}" DIRECTORY)
  install(FILES "${knurl_root}/${path}" DESTINATION "share/doc/knurl/libknurl/${directory}")
endforeach()
install(FILES "${knurl_root}/lua/knurl.lua" "${knurl_root}/lua/knurl.meta.lua"
  DESTINATION share/doc/knurl/lua)
foreach(name curl libxml2 libssh2 nghttp2 openssl zlib lua)
  install(FILES "${KNURL_SDK_ROOT}/cpkt/share/doc/cpkt/core/third_party/${name}/LICENSE"
    DESTINATION "share/doc/knurl/licenses/${name}")
endforeach()
foreach(pair "lonejson|liblonejson" "pslog|libpslog" "pid0|libpid0")
  string(REPLACE "|" ";" fields "${pair}")
  list(GET fields 0 name)
  list(GET fields 1 directory)
  install(FILES "${KNURL_SDK_ROOT}/${name}/share/doc/${directory}/LICENSE"
    DESTINATION "share/doc/knurl/licenses/${name}")
endforeach()
install(FILES "${KNURL_SDK_ROOT}/cpkt/share/doc/cpkt/LICENSE" DESTINATION share/doc/knurl/licenses/cpkt)
install(FILES "${knurl_root}/vendor/kore/LICENSE" DESTINATION share/doc/knurl/licenses/kore)
install(FILES "${knurl_root}/LICENSES/BSD-queue.txt" "${knurl_root}/LICENSES/musl.txt" DESTINATION share/doc/knurl/licenses)
file(READ "${KNURL_SDK_ROOT}/manifest.txt" dependencies)
string(APPEND dependencies "lua=5.5.1\n")
file(READ "${KNURL_SDK_ROOT}/cpkt/share/cpkt/packages/core.json" cpkt_manifest)
string(JSON component_count LENGTH "${cpkt_manifest}" components)
math(EXPR last_component "${component_count} - 1")
foreach(index RANGE ${last_component})
  string(JSON name GET "${cpkt_manifest}" components ${index} name)
  if(name MATCHES "^(curl|libxml2|libssh2|nghttp2|openssl|zlib)$")
    string(JSON version GET "${cpkt_manifest}" components ${index} version)
    string(APPEND dependencies "${name}=${version}\n")
  endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/DEPENDENCIES" "${dependencies}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/DEPENDENCIES" DESTINATION share/doc/knurl)

  find_program(KNURL_QEMU qemu-aarch64 REQUIRED)
  set(test_emulator "${KNURL_QEMU}")

configure_file(build.cmake.in binary-build.cmake @ONLY)

  "-DWORK=${CMAKE_CURRENT_BINARY_DIR}/tests/cli" "-DROOT=${knurl_root}" "-DEMULATOR=${test_emulator}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/CLI.cmake")
add_test(NAME knurl.http COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/HTTP.sh"
  "$<TARGET_FILE:knurl>" "${CMAKE_CURRENT_BINARY_DIR}/tests/http" "${test_emulator}")

  add_executable(knurl_test_orphan tests/orphan.c)
  target_compile_options(knurl_test_orphan PRIVATE -std=c89 -Wall -Wextra -Wpedantic -Werror)

  add_test(NAME knurl.scratch COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/Scratch.sh"
    "$<TARGET_FILE:knurl>" "${CMAKE_CURRENT_BINARY_DIR}/tests/scratch"
    "${CMAKE_CURRENT_BINARY_DIR}/../tests/mmdb-fixtures" "$<TARGET_FILE:knurl_test_orphan>")
  set_tests_properties(knurl.scratch PROPERTIES TIMEOUT 90)
endif()

  "-DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/Package.cmake")
