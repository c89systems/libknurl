cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_TOOLCHAIN_FILE OR NOT KNURL_TARGET OR NOT KNURL_SDK_ROOT OR NOT LONEJSON_LUA_SOURCE)
  message(FATAL_ERROR "Use make lua with the selected lifecycle toolchain and dependency prefixes")
endif()
project(lonejson_lua VERSION 0.49.0 LANGUAGES C)
set(CMAKE_C90_STANDARD_COMPILE_OPTION "-std=c89")
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/OwnedPaths.cmake")
knurl_require_owned_path("${CMAKE_BINARY_DIR}" "LoneJSON Lua build")
knurl_require_owned_path("${CMAKE_INSTALL_PREFIX}" "LoneJSON Lua SDK")
knurl_require_owned_path("${LONEJSON_LUA_SOURCE}" "LoneJSON Lua source")
foreach(name cpkt lonejson)
  list(APPEND CMAKE_PREFIX_PATH "${KNURL_SDK_ROOT}/${name}")
  list(APPEND CMAKE_FIND_ROOT_PATH "${KNURL_SDK_ROOT}/${name}")
endforeach()
find_package(Lua 5.5.1 EXACT CONFIG REQUIRED)
find_package(lonejson 0.49.0 EXACT CONFIG REQUIRED)
# The released Lua product includes the same public header beside its source.
file(SHA256 "${LONEJSON_LUA_SOURCE}/include/lonejson.h" source_header)
file(SHA256 "${KNURL_SDK_ROOT}/lonejson/include/lonejson.h" sdk_header)
if(NOT source_header STREQUAL sdk_header)
  message(FATAL_ERROR "The released LoneJSON Lua source and C SDK headers differ")
endif()
add_library(lonejson_lua SHARED "${LONEJSON_LUA_SOURCE}/src/lua/lonejson_lua.c")
set_target_properties(lonejson_lua PROPERTIES OUTPUT_NAME lonejson_core PREFIX "" SUFFIX ".so"
  EXPORT_NAME module C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
# Keep the released provider's warning contract; owned knurl code remains pedantic C89.
target_compile_options(lonejson_lua PRIVATE -std=c89 -Wall -Wextra -Werror)
target_compile_definitions(lonejson_lua PRIVATE _GNU_SOURCE=1 $<$<PLATFORM_ID:Darwin>:_DARWIN_C_SOURCE=1>)
target_include_directories(lonejson_lua PUBLIC $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(lonejson_lua PUBLIC lonejson::lonejson cpkt::lua_shared)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(lonejson_lua PRIVATE -Wl,--no-undefined)
endif()
install(TARGETS lonejson_lua EXPORT lonejsonLuaTargets LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${LONEJSON_LUA_SOURCE}/src/lua/lonejson_lua.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES "${LONEJSON_LUA_SOURCE}/lua/lonejson/init.lua" DESTINATION "${CMAKE_INSTALL_DATADIR}/lua/5.5/lonejson")
# require('lonejson.core') and C interop resolve to one shared provider instance.
file(CREATE_LINK "../../../lonejson_core.so" "${CMAKE_CURRENT_BINARY_DIR}/core.so" SYMBOLIC)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/core.so" DESTINATION "${CMAKE_INSTALL_LIBDIR}/lua/5.5/lonejson")
install(EXPORT lonejsonLuaTargets NAMESPACE lonejson_lua:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/lonejson_lua")
configure_package_config_file(Config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/lonejson_luaConfig.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/lonejson_lua")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/lonejson_luaConfigVersion.cmake"
  VERSION "${PROJECT_VERSION}" COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/lonejson_luaConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/lonejson_luaConfigVersion.cmake"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/lonejson_lua")
