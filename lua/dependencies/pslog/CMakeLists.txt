cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_TOOLCHAIN_FILE OR NOT ROTMOS_TARGET OR NOT ROTMOS_SDK_ROOT OR NOT PSLOG_LUA_SOURCE)
  message(FATAL_ERROR "Use make lua with the selected lifecycle toolchain and dependency prefixes")
endif()
project(pslog_lua VERSION 0.13.0 LANGUAGES C)
set(CMAKE_C90_STANDARD_COMPILE_OPTION "-std=c89")
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/OwnedPaths.cmake")
rotmos_require_owned_path("${CMAKE_BINARY_DIR}" "pslog Lua build")
rotmos_require_owned_path("${CMAKE_INSTALL_PREFIX}" "pslog Lua SDK")
rotmos_require_owned_path("${PSLOG_LUA_SOURCE}" "pslog Lua source")
foreach(name cpkt pslog)
  list(APPEND CMAKE_PREFIX_PATH "${ROTMOS_SDK_ROOT}/${name}")
  list(APPEND CMAKE_FIND_ROOT_PATH "${ROTMOS_SDK_ROOT}/${name}")
endforeach()
find_package(Lua 5.5.1 EXACT CONFIG REQUIRED)
find_package(pslog 0.13.0 EXACT CONFIG REQUIRED)
file(SHA256 "${PSLOG_LUA_SOURCE}/include/pslog.h" source_header)
file(SHA256 "${ROTMOS_SDK_ROOT}/pslog/include/pslog.h" sdk_header)
if(NOT source_header STREQUAL sdk_header)
  message(FATAL_ERROR "The released pslog Lua source and C SDK headers differ")
endif()
add_library(pslog_lua SHARED "${PSLOG_LUA_SOURCE}/lua/src/pslog_lua.c")
set_target_properties(pslog_lua PROPERTIES OUTPUT_NAME pslog_core PREFIX "" SUFFIX ".so"
  EXPORT_NAME module C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_options(pslog_lua PRIVATE -std=c89 -Wall -Wextra -Werror)
# C89 permits inline as an ordinary macro name. Drop the released binding's
# optional C99 static-inline hint without modifying its source distribution.
target_compile_definitions(pslog_lua PRIVATE inline= _GNU_SOURCE=1 $<$<PLATFORM_ID:Darwin>:_DARWIN_C_SOURCE=1>)
target_include_directories(pslog_lua PRIVATE "${PSLOG_LUA_SOURCE}/include" PUBLIC $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(pslog_lua PUBLIC pslog::pslog_shared cpkt::lua_shared)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(pslog_lua PRIVATE -Wl,--no-undefined)
endif()
install(TARGETS pslog_lua EXPORT pslogLuaTargets LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${PSLOG_LUA_SOURCE}/include/pslog_lua.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(FILES "${PSLOG_LUA_SOURCE}/lua/pslog/init.lua" DESTINATION "${CMAKE_INSTALL_DATADIR}/lua/5.5/pslog")
file(CREATE_LINK "../../../pslog_core.so" "${CMAKE_CURRENT_BINARY_DIR}/core.so" SYMBOLIC)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/core.so" DESTINATION "${CMAKE_INSTALL_LIBDIR}/lua/5.5/pslog")
install(EXPORT pslogLuaTargets NAMESPACE pslog_lua:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/pslog_lua")
configure_package_config_file(Config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/pslog_luaConfig.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/pslog_lua")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/pslog_luaConfigVersion.cmake"
  VERSION "${PROJECT_VERSION}" COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/pslog_luaConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/pslog_luaConfigVersion.cmake"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/pslog_lua")
